document.addEventListener('DOMContentLoaded', () => {
// Κατάσταση Εφαρμογής
let items = JSON.parse(localStorage.getItem('supermarket_items')) || [];
let budgetLimit = parseFloat(localStorage.getItem('supermarket_budget')) || '';
let hideChecked = JSON.parse(localStorage.getItem('supermarket_hide_checked')) || false;
let darkMode = JSON.parse(localStorage.getItem('supermarket_dark')) || false;
let wasOverBudget = false;
// Αναφορές DOM
const themeToggle = document.getElementById('theme-toggle');
const addForm = document.getElementById('add-form');
const itemNameInput = document.getElementById('item-name');
const itemQtyInput = document.getElementById('item-qty');

const itemCatInput = document.getElementById('item-category');

const listContainer = document.getElementById('list-container');
const progressText = document.getElementById('progress-text');

const progressFill = document.getElementById('progress-fill');
const rewardMsg = document.getElementById('reward-msg');
const totalCostEl = document.getElementById('total-cost');
const budgetInput = document.getElementById('budget-limit');
const remainingText = document.getElementById('remaining-text');
const bottomBar = document.getElementById('bottom-bar');
// Δόνηση κινητού
function vibrate(ms = 25) {
if (navigator.vibrate) {
navigator.vibrate(ms);
}
}
// Δημιουργία Φθινοπωρινών Φύλλων
function initLeaves() {
const container = document.getElementById('leaves-container');
const emojis = ['🍂', '🍁'];
for (let i = 0; i < 14; i++) {
const leaf = document.createElement('div');
leaf.className = 'leaf';
leaf.textContent = emojis[i % emojis.length];
leaf.style.left = Math.random() * 100 + 'vw';
leaf.style.animationDuration = (9 + Math.random() * 10) + 's';
leaf.style.animationDelay = (Math.random() * 8) + 's';
leaf.style.fontSize = (1.1 + Math.random() * 0.7) + 'rem';
container.appendChild(leaf);
}
}
// Dark Mode
function applyTheme() {
if (darkMode) {
document.body.classList.add('dark-mode');
themeToggle.textContent = '☀️';
} else {
document.body.classList.remove('dark-mode');
themeToggle.textContent = '🌙';
}
}
themeToggle.addEventListener('click', () => {
darkMode = !darkMode;
localStorage.setItem('supermarket_dark', JSON.stringify(darkMode));
applyTheme();
vibrate(20);
});
// Αποθήκευση
function saveState() {
localStorage.setItem('supermarket_items', JSON.stringify(items));
localStorage.setItem('supermarket_budget', budgetLimit);
localStorage.setItem('supermarket_hide_checked', JSON.stringify(hideChecked));
}
// Προσθήκη Προϊόντος
function addItem(name, qty, price, category) {
const cleanName = name.trim();
if (!cleanName) return;
// Αν υπάρχει ήδη ίδιο όνομα & κατηγορία, αύξηση ποσότητας
const existing = items.find(i => i.name.toLowerCase() === cleanName.toLowerCase() && i.category === category);
if (existing) {
existing.qty += qty;
if (price > 0) existing.price = price;
existing.checked = false;
} else {
items.push({
id: Date.now().toString() + Math.random().toString(36).slice(2, 6),
name: cleanName,
qty: qty,
price: price,
category: category,
checked: false
});
}
vibrate(30);
saveState();
render();
}
addForm.addEventListener('submit', (e) => {
e.preventDefault();
const name = itemNameInput.value;
const qty = Math.max(1, parseInt(itemQtyInput.value) || 1);
const price = Math.max(0, parseFloat(itemPriceInput.value) || 0);
const category = itemCatInput.value;
addItem(name, qty, price, category);
itemNameInput.value = '';
itemQtyInput.value = '1';
itemPriceInput.value = '';
itemNameInput.focus();
});
// Γρήγορα Κουμπιά 1 Πατήματος
document.querySelectorAll('.quick-chip').forEach(btn => {
btn.addEventListener('click', () => {
const name = btn.dataset.name;
const cat = btn.dataset.cat;
const price = parseFloat(btn.dataset.price) || 0;
addItem(name, 1, price, cat);
});
});
// Αναζήτηση & Απόκρυψη Αγορασμένων
searchInput.addEventListener('input', render);
toggleHideBtn.addEventListener('click', () => {
hideChecked = !hideChecked;
toggleHideBtn.classList.toggle('active', hideChecked);
saveState();
render();
vibrate(20);
});

saveState();
updateTotals();
});
// Ενημέρωση Προόδου & Κόστους
function updateTotals() {
const totalItems = items.length;
const checkedItems = items.filter(i => i.checked).length;
const percent = totalItems === 0 ? 0 : Math.round((checkedItems / totalItems) * 100);
progressText.textContent = Αγοράστηκαν ${checkedItems} / ${totalItems};
progressPercent.textContent = ${percent}%;
progressFill.style.width = ${percent}%;
if (totalItems > 0 && checkedItems === totalItems) {
progressFill.classList.add('completed');
rewardMsg.style.display = 'block';
} else {
progressFill.classList.remove('completed');
rewardMsg.style.display = 'none';
}
const totalCost = items.reduce((sum, item) => sum + (item.qty * item.price), 0);
totalCostEl.textContent = ${totalCost.toFixed(2)} €;
if (budgetLimit !== '' && budgetLimit > 0) {
const diff = budgetLimit - totalCost;
if (diff < 0) {
remainingText.textContent = Υπέρβαση: ${Math.abs(diff).toFixed(2)} €;
bottomBar.classList.add('over-budget');
if (!wasOverBudget) {
vibrate([60, 40, 60]);
wasOverBudget = true;
}
} else {
remainingText.textContent = Υπόλοιπο: ${diff.toFixed(2)} €;
bottomBar.classList.remove('over-budget');
wasOverBudget = false;
}
} else {
remainingText.textContent = 'Χωρίς όριο';
bottomBar.classList.remove('over-budget');
wasOverBudget = false;
}
}
// Εμφάνιση Λίστας
function render() {
toggleHideBtn.classList.toggle('active', hideChecked);
const query = searchInput.value.trim().toLowerCase();
const filtered = items.filter(item => {
if (hideChecked && item.checked) return false;
if (query && !item.name.toLowerCase().includes(query)) return false;
return true;
});
listContainer.innerHTML = '';
if (filtered.length === 0) {
listContainer.innerHTML = <div class="empty-state">${items.length === 0 ? 'Το καλάθι σου είναι άδειο. Πρόσθεσε προϊόντα παραπάνω!' : 'Δεν βρέθηκαν προϊόντα με αυτά τα κριτήρια.'}</div>;
updateTotals();
return;
}
// Ομαδοποίηση ανά κατηγορία
const groups = {};
filtered.forEach(item => {
if (!groups[item.category]) groups[item.category] = [];
groups[item.category].push(item);
});
Object.keys(groups).forEach(category => {
const groupDiv = document.createElement('div');
groupDiv.className = 'category-group';
const titleDiv = document.createElement('div');
titleDiv.className = 'category-title';
titleDiv.textContent = category;
groupDiv.appendChild(titleDiv);
groups[category].forEach(item => {
const card = document.createElement('div');
card.className = item-card ${item.checked ? 'checked' : ''};
card.innerHTML = <div class="item-left"> <input type="checkbox" class="item-checkbox" ${item.checked ? 'checked' : ''}> <span class="item-title">${escapeHTML(item.name)}</span> </div> <div class="item-controls"> <div class="qty-box"> <button type="button" class="qty-btn minus">−</button> <span class="qty-val">${item.qty}</span> <button type="button" class="qty-btn plus">+</button> </div> <div class="inline-price-wrap" title="Τιμή μονάδας €"> <input type="number" class="inline-price" value="${item.price > 0 ? item.price : ''}" placeholder="0.00" min="0" step="0.01"> <span style="font-size:0.8rem;color:var(--text-muted)">€</span> </div> <button type="button" class="del-btn" title="Διαγραφή">✕</button> </div>;
// Events στοιχείου
const checkbox = card.querySelector('.item-checkbox');
checkbox.addEventListener('change', () => {
item.checked = checkbox.checked;
vibrate(25);
saveState();
render();
});
card.querySelector('.minus').addEventListener('click', () => {
if (item.qty > 1) {
item.qty--;
vibrate(15);
saveState();
render();
}
});
card.querySelector('.plus').addEventListener('click', () => {
item.qty++;
vibrate(15);
saveState();
render();
});
const priceInput = card.querySelector('.inline-price');
priceInput.addEventListener('change', () => {
item.price = Math.max(0, parseFloat(priceInput.value) || 0);
saveState();
updateTotals();
});
card.querySelector('.del-btn').addEventListener('click', () => {
items = items.filter(i => i.id !== item.id);
vibrate(30);
saveState();
render();
});
groupDiv.appendChild(card);
});
listContainer.appendChild(groupDiv);
});
updateTotals();
}
// Ασφάλεια HTML χαρακτήρων
function escapeHTML(str) {
return str.replace(/[&<>'"]/g, tag => ({
'&': '&', '<': '<', '>': '>', "'": ''', '"': '"'
}[tag]));
}
// Σβήσιμο Αγορασμένων
document.getElementById('clear-checked-btn').addEventListener('click', () => {
if (!items.some(i => i.checked)) return;
items = items.filter(i => !i.checked);
vibrate(35);
saveState();
render();
});
// Καθαρισμός Όλης της Λίστας
document.getElementById('clear-all-btn').addEventListener('click', () => {
if (items.length === 0) return;
if (confirm('Θέλεις σίγουρα να διαγράψεις όλα τα προϊόντα της λίστας;')) {
items = [];
vibrate(50);
saveState();
render();
}
});
// Κοινοποίηση Λίστας (Viber / WhatsApp / Clipboard)
document.getElementById('share-btn').addEventListener('click', async () => {
if (items.length === 0) {
alert('Η λίστα είναι άδεια!');
return;
}
let text = '🛒 Λίστα Σούπερ Μάρκετ\n\n';
items.forEach(item => {
const mark = item.checked ? '✅' : '⬜';
const priceStr = item.price > 0 ?  (${(item.qty * item.price).toFixed(2)}€) : '';
text += ${mark} ${item.name} x${item.qty}${priceStr}\n;
});
const total = items.reduce((s, i) => s + i.qty * i.price, 0);
text += \n💰 *Σύνολο:* ${total.toFixed(2)} €;
if (navigator.share) {
try {
await navigator.share({ title: 'Λίστα Σούπερ Μάρκετ', text });
} catch (err) {}
} else {
navigator.clipboard.writeText(text);
alert('Η λίστα αντιγράφηκε στο πρόχειρο! Μπορείς να την κάνεις επικόλληση στο Viber ή WhatsApp.');
}
vibrate(25);
});
// Εκκίνηση
applyTheme();
initLeaves();
render();
});
